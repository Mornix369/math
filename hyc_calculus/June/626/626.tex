\section*{Unit 1: Limits and Continuity}

\begin{enumerate}[1.]
    \item Evaluate the limit: $\displaystyle \lim_{x \to 2} \frac{x^2 - 4}{x^2 - x - 2}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $\dfrac{4}{3}$ \qquad C. $1$ \qquad D. The limit does not exist
    \end{enumerate}

    \item Find the equation of the horizontal asymptote for the rational function: $f(x) = \dfrac{3x^3 - 5x + 2}{1 - 2x^3}$
    \begin{enumerate}[A.]
        \item $y = 3$ \qquad B. $y = 0$ \qquad C. $y = -\dfrac{3}{2}$ \qquad D. No horizontal asymptote
    \end{enumerate}

    \item Given the piecewise function $f(x) = \begin{cases} 
      kx + 3, & x \le 3 \\ 
      k^2x - 9, & x > 3 
   \end{cases}$ is continuous at $x = 3$, the possible values for the real number $k$ are:
    \begin{enumerate}[A.]
        \item $k = 2$ or $k = -2$ \qquad B. $k = 2$ or $k = -1$ \qquad C. $k = -2$ or $k = 1$ \qquad D. $k = 3$ or $k = -1$
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to 0} \frac{\sin(4x)}{3x}$
    \begin{enumerate}[A.]
        \item $\dfrac{4}{3}$ \qquad B. $\dfrac{3}{4}$ \qquad C. $0$ \qquad D. $1$
    \end{enumerate}

    \item For the function $f(x) = \dfrac{x-1}{|x-1|}$, which of the following statements about $\displaystyle \lim_{x \to 1} f(x)$ is true?
    \begin{enumerate}[A.]
        \item The limit is $1$ \qquad B. The limit is $-1$ \qquad C. The limit is $0$ \qquad D. The limit does not exist because the left and right limits are not equal
    \end{enumerate}

    \item At which of the following points does the function $f(x) = \dfrac{x+2}{x^2 - 4}$ have a vertical asymptote?
    \begin{enumerate}[A.]
        \item Only at $x = 2$ \qquad B. Only at $x = -2$ \qquad C. At $x = 2$ and $x = -2$ \qquad D. No vertical asymptote exists
    \end{enumerate}

    \item Intermediate Value Theorem on a closed interval. Let $f(x)$ be continuous on the closed interval $[1, 5]$, with $f(1) = -2$ and $f(5) = 6$. Which of the following values of $M$ is NOT guaranteed by the theorem to have a value $c$ in the interval $(1, 5)$ such that $f(c) = M$?
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $3$ \qquad C. $-1$ \qquad D. $7$
    \end{enumerate}
\end{enumerate}

\section*{Unit 2: Differentiation: Definition and Fundamental Properties}

\begin{enumerate}[1.]
    \setcounter{enumi}{7}
    \item The limit expression $\displaystyle \lim_{h \to 0} \frac{(2+h)^4 - 16}{h}$ represents the derivative of which of the following functions at the specified point?
    \begin{enumerate}[A.]
        \item $f(x) = x^4$ at $x = 2$ \qquad B. $f(x) = x^4$ at $x = 16$ \\ C. $f(x) = (x+2)^4$ at $x = 0$ \qquad D. $f(x) = 16$ at $x = 2$
    \end{enumerate}

    \item If $f(x) = x^3 - 2x$, find the instantaneous rate of change of the function at $x = 2$.
    \begin{enumerate}[A.]
        \item $4$ \qquad B. $10$ \qquad C. $12$ \qquad D. $14$
    \end{enumerate}

    \item Which of the following statements about the relationship between differentiability and continuity is true?
    \begin{enumerate}[A.]
        \item A continuous function is always differentiable everywhere. \qquad B. A discontinuous function can be differentiable at some points. \\ C. A differentiable function is always continuous. \qquad D. If a function is not differentiable at a point, it must be discontinuous at that point.
    \end{enumerate}

    \item Let $f(x) = (3x^2 + 1)(2x - 5)$. Find its derivative $f'(x)$.
    \begin{enumerate}[A.]
        \item $f'(x) = 18x^2 - 30x + 2$ \qquad B. $f'(x) = 6x$ \\ C. $f'(x) = 12x^2 - 30x$ \qquad D. $f'(x) = 18x^2 - 30x$
    \end{enumerate}

    \item Given $f(x) = \dfrac{x^2 + 3}{x - 1}$, find the value of $f'(2)$.
    \begin{enumerate}[A.]
        \item $-3$ \qquad B. $-1$ \qquad C. $1$ \qquad D. $3$
    \end{enumerate}

    \item Find the slope of the tangent line to the curve $y = 2\sin x + \cos x$ at the point $(0, 1)$.
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $1$ \qquad C. $2$ \qquad D. $-1$
    \end{enumerate}

    \item Let $f(x) = \dfrac{4}{\sqrt{x}} - 3x^2$. Find $f'(1)$.
    \begin{enumerate}[A.]
        \item $-8$ \qquad B. $-4$ \qquad C. $-2$ \qquad D. $2$
    \end{enumerate}

    \item The position of an object on the closed interval $[0, 3]$ is given by $s(t) = t^2 + 2t$. What is the average velocity of the object over this time interval?
    \begin{enumerate}[A.]
        \item $5$ \qquad B. $8$ \qquad C. $15$ \qquad D. $3$
    \end{enumerate}
\end{enumerate}

\section*{Unit 3: Differentiation: Composite, Implicit, and Inverse Functions}

\begin{enumerate}[1.]
    \setcounter{enumi}{15}
    \item Let $f(x) = \ln(x^2 + 3x)$. Find $f'(x)$.
    \begin{enumerate}[A.]
        \item $\dfrac{1}{x^2 + 3x}$ \qquad B. $\dfrac{2x + 3}{x^2 + 3x}$ \qquad C. $\dfrac{2x}{x^2 + 3x}$ \qquad D. $\dfrac{1}{2x+3}$
    \end{enumerate}

    \item Given $y = e^{\sin x}$, find $\dfrac{dy}{dx}$.
    \begin{enumerate}[A.]
        \item $e^{\sin x}$ \qquad B. $\cos x \cdot e^{\sin x}$ \qquad C. $-\cos x \cdot e^{\sin x}$ \qquad D. $\sin x \cdot e^{\sin x - 1}$
    \end{enumerate}

    \item For the implicit function determined by the equation $x^2 + xy + y^2 = 7$, the derivative $\dfrac{dy}{dx}$ at the point $(1, 2)$ is:
    \begin{enumerate}[A.]
        \item $-\dfrac{4}{5}$ \qquad B. $-\dfrac{5}{4}$ \qquad C. $-\dfrac{2}{5}$ \qquad D. $1$
    \end{enumerate}

    \item Let $f(x) = \arctan(2x)$. Find $f'(x)$.
    \begin{enumerate}[A.]
        \item $\dfrac{2}{1 + 4x^2}$ \qquad B. $\dfrac{1}{1 + 4x^2}$ \qquad C. $\dfrac{2}{1 + x^2}$ \qquad D. $\dfrac{1}{1 + 2x^2}$
    \end{enumerate}

    \item Let $g(x)$ be the inverse function of $f(x)$. If $f(3) = 5$ and $f'(3) = 4$, then the value of $g'(5)$ is:
    \begin{enumerate}[A.]
        \item $\dfrac{1}{5}$ \qquad B. $\dfrac{1}{3}$ \qquad C. $\dfrac{1}{4}$ \qquad D. $4$
    \end{enumerate}

    \item Let $y = \cos^3(2x)$. Find $\dfrac{dy}{dx}$.
    \begin{enumerate}[A.]
        \item $-3\cos^2(2x)\sin(2x)$ \qquad B. $-6\cos^2(2x)\sin(2x)$ \\ C. $6\cos^2(2x)\sin(2x)$ \qquad D. $-6\sin^2(2x)$
    \end{enumerate}

    \item For the curve determined by the equation $x^2 - y^2 = 9$, find its second derivative $\dfrac{d^2y}{dx^2}$ at any point.
    \begin{enumerate}[A.]
        \item $\dfrac{1}{y}$ \qquad B. $\dfrac{x^2}{y^3}$ \qquad C. $-\dfrac{9}{y^3}$ \qquad D. $\dfrac{9}{y^3}$
    \end{enumerate}
\end{enumerate}

\section*{Unit 4: Contextual Applications of Differentiation}

\begin{enumerate}[1.]
    \setcounter{enumi}{22}
    \item A particle moves along a straight line with its position function given by $s(t) = 2t^3 - 9t^2 + 12t - 4$ (for $t \ge 0$). During which of the following time intervals is the particle moving to the left (in the negative direction)?
    \begin{enumerate}[A.]
        \item $(0, 1)$ \qquad B. $(1, 2)$ \qquad C. $(2, \infty)$ \qquad D. The particle never moves to the left
    \end{enumerate}

    \item Following the previous question, what is the acceleration $a(2)$ of the particle at $t = 2$?
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $3$ \qquad C. $6$ \qquad D. $12$
    \end{enumerate}

    \item (Related Rates) The side length of a square is increasing at a constant rate of $3\text{ cm/s}$. What is the rate of increase of its area when the side length is $10\text{ cm}$?
    \begin{enumerate}[A.]
        \item $30\text{ cm}^2/\text{s}$ \qquad B. $60\text{ cm}^2/\text{s}$ \qquad C. $90\text{ cm}^2/\text{s}$ \qquad D. $100\text{ cm}^2/\text{s}$
    \end{enumerate}

    \item (Related Rates) A ladder $5\text{ m}$ long is leaning against a vertical wall. If the base of the ladder is pulled away from the wall at a speed of $1.5\text{ m/s}$, what is the downward speed of the top of the ladder when the base is $3\text{ m}$ from the wall?
    \begin{enumerate}[A.]
        \item $1.125\text{ m/s}$ \qquad B. $2.0\text{ m/s}$ \qquad C. $1.5\text{ m/s}$ \qquad D. $0.875\text{ m/s}$
    \end{enumerate}

    \item Use the linear approximation of the function $f(x) = \sqrt{x}$ at $x = 4$ to estimate the value of $\sqrt{4.2}$.
    \begin{enumerate}[A.]
        \item $2.025$ \qquad B. $2.05$ \qquad C. $2.1$ \qquad D. $2.0125$
    \end{enumerate}

    \item Evaluate the limit using L'H\^{o}pital's Rule: $\displaystyle \lim_{x \to 0} \frac{e^{3x} - 1}{\sin(2x)}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $1$ \qquad C. $\dfrac{2}{3}$ \qquad D. $\dfrac{3}{2}$
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to \infty} \frac{\ln(x)}{x^2}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $1$ \qquad C. $\infty$ \qquad D. The limit does not exist
    \end{enumerate}

    \item A cylindrical water tank has a constant radius and a base area of $25\pi\text{ m}^2$. Water is being pumped into the tank at a constant rate of $5\text{ m}^3/\text{min}$. What is the rate at which the water level $h$ is rising?
    \begin{enumerate}[A.]
        \item $\dfrac{1}{5}\text{ m/min}$ \qquad B. $\dfrac{1}{5\pi}\text{ m/min}$ \qquad C. $5\text{ m/min}$ \qquad D. $25\pi\text{ m/min}$
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to 0} \frac{\sin(5x)}{x}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $1$ \qquad C. $5$ \qquad D. The limit does not exist
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to \infty} \frac{e^x}{x^3}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $1$ \qquad C. $\dfrac{1}{3}$ \qquad D. $\infty$
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to 2} \frac{x^2 - 4x + 4}{x^3 - 8}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $\dfrac{1}{12}$ \qquad C. $\dfrac{1}{4}$ \qquad D. The limit does not exist
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to 0} \frac{1 - \cos x}{x^2}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $\dfrac{1}{2}$ \qquad C. $1$ \qquad D. $\infty$
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to 1} \frac{\ln x}{x - 1}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $1$ \qquad C. $e$ \qquad D. The limit does not exist
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to 0} \frac{e^x - 1 - x}{x^2}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $\dfrac{1}{2}$ \qquad C. $1$ \qquad D. $\infty$
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to \infty} \frac{\ln(x^5)}{x}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $1$ \qquad C. $5$ \qquad D. $\infty$
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to \pi} \frac{\sin x}{x - \pi}$
    \begin{enumerate}[A.]
        \item $1$ \qquad B. $0$ \qquad C. $-1$ \qquad D. The limit does not exist
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to 0} \frac{x \cos x}{\sin(2x)}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $\dfrac{1}{2}$ \qquad C. $1$ \qquad D. $2$
    \end{enumerate}

    \item Evaluate the limit: $\displaystyle \lim_{x \to \infty} \frac{3x^2 - 5x + 7}{4x^2 + 1}$
    \begin{enumerate}[A.]
        \item $0$ \qquad B. $\dfrac{3}{4}$ \qquad C. $3$ \qquad D. $\infty$
    \end{enumerate}

\end{enumerate}
